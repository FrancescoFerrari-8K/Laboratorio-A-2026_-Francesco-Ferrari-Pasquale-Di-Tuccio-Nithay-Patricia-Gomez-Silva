//cercaFilm: questo metodo permette la ricerca di un film nel file proiezioni.csv
	//il metodo permette la ricerca di un film secondo diversi parametri specificati da un intero 'modalità' compreso tra 0 e 7, che specifica in quale colonna del file proiezioni
        //effettuare la ricerca e una chiave di ricerca 'chiave' di tipo String, che specifica la caratteristica ricercata dall'utente (es. tirolo, giorno e ora ecc..).
	//Questo metodo utilizza il metodo caricaFilm() della classe Film per l'accesso al file proiezioni.csv e ne separa le colonne, contenute nell'Array di Linked list restiuito da  
        //cerfaFilm().
	//Questo metodo restituisce una linked list di oggetti di tipo Film corrispondenti alle occorrenze della stringa nel file proiezioni.csv cercata secondo la modalità specificata



public LinkedList<Film> cercaFilm(int modalità, String chiave){
		LinkedList<Film> ListaFilm= new LinkedList<Film>();//inizializzo la linked list di oggetti film che soddisfano la chiave di ricerca
		try {
			ArrayList<LinkedList<String>> dati=Film.caricaFilm();//carico i film presenti in proiezioni.csv
			//separo le colonne per creare gli oggetti film da aggiungere alla linked List di oggetti Film che soddisfano la chiave di ricerca ListaFilm
			LinkedList<String> dataeora=dati.get(0);
			LinkedList<String> titolo=dati.get(1);
			LinkedList<String> genere=dati.get(2);
			LinkedList<String> regista=dati.get(3);
			LinkedList<String> anno=dati.get(4);
			LinkedList<String> durata=dati.get(5);
			LinkedList<String> età=dati.get(6);
			LinkedList<String> prezzo=dati.get(7);
			LinkedList<String> searchable=dati.get(modalità);//cerco nella colonna specificata dalla modalità
			for(int i=1; i<dati.size(); i++) //salto il primo elemento perchè è l'intestazione della colonna! Scandisco la lista (n confronti)
				if(searchable.get(i).equals(chiave)) {//se l'elemento i esimo della colonna corrisponde alla chiave di ricerca lo aggiungo alla lista!
					Film f=new Film(dataeora.get(i),titolo.get(i),genere.get(i),regista.get(i),anno.get(i),durata.get(i),età.get(i),prezzo.get(i));
					ListaFilm.add(f);
				}
		} catch (IOException e) {
			// TODO Auto-generated catch block
			e.printStackTrace();
		}
		
		return ListaFilm;
		
	}
